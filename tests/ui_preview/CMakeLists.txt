cmake_minimum_required(VERSION 3.16)
project(th2822_ui_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl EXCLUDE_FROM_ALL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
add_executable(panel_preview main.cpp ../../main/ui.cpp ../../core/meter.cpp ../../core/i18n.cpp ../../core/i18n_generated.cpp ../../fonts/panel_cjk_14.c ../../fonts/panel_cjk_20.c ../../fonts/panel_cjk_24.c)
target_include_directories(panel_preview PRIVATE ../../main ../../core .. .)
target_link_libraries(panel_preview PRIVATE lvgl PkgConfig::SDL2)
